cmake_minimum_required(VERSION 3.16)
project(multigroup_solver_prototype LANGUAGES CXX)
include(CTest)
option(MGSOLVE_SANITIZE "Enable address/undefined sanitizers" OFF)
add_library(mgsolve INTERFACE)
target_include_directories(mgsolve INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/../../src/radiation/nested)
target_compile_features(mgsolve INTERFACE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mgsolve INTERFACE -UNDEBUG -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  if(MGSOLVE_SANITIZE)
    target_compile_options(mgsolve INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mgsolve INTERFACE -fsanitize=address,undefined)
  endif()
elseif(MSVC)
  target_compile_options(mgsolve INTERFACE /W4 /fp:strict)
endif()
add_executable(unit_tests unit_tests.cpp)
target_link_libraries(unit_tests PRIVATE mgsolve)
add_test(NAME unit_tests COMMAND unit_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/case_driver.cpp")
  add_executable(case_driver case_driver.cpp)
  target_link_libraries(case_driver PRIVATE mgsolve)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME high_precision_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/reference_validation.py $<TARGET_FILE:case_driver> --output ${CMAKE_CURRENT_BINARY_DIR}/reference_report.json)
  endif()
endif()
if(Python3_FOUND)
  foreach(audit inner certificate width)
    add_executable(independent_${audit}_probe independent_${audit}_probe.cpp)
    target_link_libraries(independent_${audit}_probe PRIVATE mgsolve)
    add_test(NAME independent_${audit}_audit COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/independent_${audit}_audit.py $<TARGET_FILE:independent_${audit}_probe>)
  endforeach()
endif()

add_executable(independent_reduction_audit independent_reduction_audit.cpp)
target_link_libraries(independent_reduction_audit PRIVATE mgsolve)
add_test(NAME independent_reduction_audit COMMAND independent_reduction_audit)
